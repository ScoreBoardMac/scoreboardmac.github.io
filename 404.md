{%- comment -%}
  GitHub Pages отдаёт этот файл для любого несуществующего адреса.
  Блок на нужном языке выбирается скриптом по первому сегменту адреса (/ru/... → ru),
  остальные скрываются. Без JS видны все блоки. Блок для нового языка — по образцу ниже,
  data-lang должен совпадать с code в _data/i18n.yml.
{%- endcomment -%}

{%- comment -%}
  Меню, ссылка на главную и описание в подвале для каждого языка —
  собираются так же, как в _includes/header.html и footer.html.
{%- endcomment -%}
<script>
  var siteChrome = {
  {%- for l in site.data.i18n.languages %}
    {%- assign home = site.pages | where: "ref", "home" | where: "lang", l.code | first %}
    {{ l.code | jsonify }}: {
      home: {{ home.url | default: '/' | relative_url | jsonify }},
      footer: {{ site.data.i18n.description[l.code] | default: site.description | strip | jsonify }},
      nav: [
      {%- for path in site.minima.nav_pages -%}
        {%- assign base = site.pages | where: "path", path | first -%}
        {%- assign target = site.pages | where: "ref", base.ref | where: "lang", l.code | first | default: base -%}
        { url: {{ target.url | relative_url | jsonify }}, title: {{ target.nav_title | default: target.title | jsonify }} }{% unless forloop.last %},{% endunless %}
      {%- endfor %}
      ]
    }{% unless forloop.last %},{% endunless %}
  {%- endfor %}
  };

  // Подвал идёт после контента, поэтому ждём загрузки всей страницы.
  document.addEventListener('DOMContentLoaded', function () {
    var blocks = document.querySelectorAll('.not-found[data-lang]');
    var prefix = location.pathname.split('/')[1].toLowerCase();
    var current = document.querySelector('.not-found[data-lang="' + prefix + '"]') ||
                  document.querySelector('.not-found[data-lang="{{ site.lang }}"]');
    var lang = current.getAttribute('data-lang');
    for (var i = 0; i < blocks.length; i++) {
      if (blocks[i] !== current) blocks[i].hidden = true;
    }
    document.title = current.getAttribute('data-title') + ' | {{ site.title }}';
    if (lang === '{{ site.lang }}' || !siteChrome[lang]) return;

    // Шапка и подвал собраны на языке по умолчанию — переводим их на язык адреса.
    var chrome = siteChrome[lang];
    document.documentElement.lang = lang;
    var title = document.querySelector('.site-title');
    if (title) title.href = chrome.home;
    var items = document.querySelectorAll('.nav-items a.nav-item');
    for (var j = 0; j < items.length && j < chrome.nav.length; j++) {
      items[j].href = chrome.nav[j].url;
      items[j].textContent = chrome.nav[j].title;
    }
    var footer = document.querySelector('.site-footer .footer-col p');
    if (footer) footer.textContent = chrome.footer;
  });
</script>
